    <script>

        const API_URL = "https://ai-hackathon-project-ozob.onrender.com/ask-math";


        const chatContainer = document.getElementById('chatContainer');

        const userInput = document.getElementById('userInput');

        const sendBtn = document.getElementById('sendBtn');


        async function sendMessage() {

            const text = userInput.value.trim();

            if (!text) return;


            appendMessage(text, 'user');

            userInput.value = '';

            

            const loadingDiv = appendMessage('Думаю... Ойлонуп жатам... 🤖', 'bot loading');

            chatContainer.scrollTop = chatContainer.scrollHeight;


            try {

                const response = await fetch(API_URL, {

                    method: 'POST',

                    headers: {

                        'Content-Type': 'application/json'

                    },

                    body: JSON.stringify({ question: text })

                });


                const data = await response.json();

                

                loadingDiv.remove();


                if (data.status === 'success') {

                    appendMessage(data.answer, 'bot');

                } else {

                    appendMessage('Извини, произошла ошибка на сервере. 😢', 'bot');

                }


            } catch (error) {

                console.error(error);

                loadingDiv.remove();

                appendMessage('Не удалось связаться с сервером. Проверь интернет. 🌐', 'bot');

            }


            chatContainer.scrollTop = chatContainer.scrollHeight;

        }


        function appendMessage(text, className) {

            const messageDiv = document.createElement('div');

            messageDiv.className = `message ${className}`;

            messageDiv.textContent = text;

            chatContainer.appendChild(messageDiv);

            return messageDiv;

        }


        userInput.addEventListener('keydown', (e) => {

            if (e.key === 'Enter' && !e.shiftKey) {

                e.preventDefault();

                sendMessage();

            }

        });

    </script>
